#include <iostream>
#include <string>
using namespace std;

class Course {
private:
    string course_code;
    string course_name;
public:
    Course(string code = "", string name = "") : course_code(code), course_name(name) {}
    void set_course_code(string code) { course_code = code; }
    void set_course_name(string name) { course_name = name; }
    string get_course_code() const { return course_code; }
    string get_course_name() const { return course_name; }
};

class Grade {
private:
    int mark;
    char the_grade;
    bool is_set;
public:
    Grade(int m = 0) : mark(m), is_set(false) {
        if (m >= 0 && m <= 100) {
            the_grade = calculate_grade();
            is_set = true;
        } else {
            the_grade = 'E';
        }
    }
    void set_mark(int m) {
        if (!is_set && m >= 0 && m <= 100) {
            mark = m;
            the_grade = calculate_grade();
            is_set = true;
        }
    }
    int get_mark() const { return mark; }
    char get_grade() const { return the_grade; }
    bool is_grade_set() const { return is_set; }
private:
    char calculate_grade() const {
        if (mark > 69) return 'A';
        if (mark > 59) return 'B';
        if (mark > 49) return 'C';
        if (mark > 39) return 'D';
        return 'E';
    }
};

class Student {
private:
    string reg_number;
    string name;
    int age;
    Course course;
    Grade grade;
public:
    Student(string reg = "", string n = "", int a = 0, Course c = Course(), Grade g = Grade()) 
        : reg_number(reg.substr(0, 20)), name(n.substr(0, 50)), age(a), course(c), grade(g) {}
    void set_regNo(string reg) { reg_number = reg.substr(0, 20); }
    void set_name(string n) { name = n.substr(0, 50); }
    void set_age(int a) { age = a; }
    void set_course(Course c) { course = c; }
    void set_grade(Grade g) { grade = g; }
    void set_marks(int m) { grade.set_mark(m); }
    string get_regNo() const { return reg_number; }
    string get_name() const { return name; }
    int get_age() const { return age; }
    Course get_course() const { return course; }
    Grade get_grade() const { return grade; }
    int get_marks() const { return grade.get_mark(); }
    void calculate_grade() {
        if (!grade.is_grade_set()) {
            grade.set_mark(grade.get_mark());
        } else {
            cout << "Grade for " << reg_number << " is already set and cannot be modified." << endl;
        }
    }
};

class List {
private:
    static const int MAX_SIZE = 40;
    Student students[MAX_SIZE];
    int size;
public:
    List() : size(0) {}
    List(const Student& s) : size(1) {
        students[0] = s;
    }
    List(const List& other) : size(other.size) {
        for (int i = 0; i < size; i++) {
            students[i] = other.students[i];
        }
    }
    void add_student(const Student& s) {
        if (size >= MAX_SIZE) {
            cout << "List is full!" << endl;
            return;
        }
        students[size] = s;
        size++;
        cout << "Student " << s.get_regNo() << " added." << endl;
    }
    void add_student_at(const Student& s, int position) {
        if (position < 0 || position > size || size >= MAX_SIZE) {
            cout << "Invalid position or list is full!" << endl;
            return;
        }
        for (int i = size; i > position; i--) {
            students[i] = students[i - 1];
        }
        students[position] = s;
        size++;
        cout << "Student " << s.get_regNo() << " added at position " << position << "." << endl;
    }
    void remove_student(const Student& s) {
        for (int i = 0; i < size; i++) {
            if (students[i].get_regNo() == s.get_regNo()) {
                remove_student_at(i);
                return;
            }
        }
        cout << "Student " << s.get_regNo() << " not found!" << endl;
    }
    void remove_student_at(int position) {
        if (position < 0 || position >= size) {
            cout << "Invalid position!" << endl;
            return;
        }
        for (int i = position; i < size - 1; i++) {
            students[i] = students[i + 1];
        }
        size--;
        cout << "Student at position " << position << " removed." << endl;
    }
    int get_size() const {
        return size;
    }
    void destroy() {
        size = 0;
        cout << "List cleared." << endl;
    }
    Student& get_student(int index) {
        if (index < 0 || index >= size) {
            throw "Invalid index!";
        }
        return students[index];
    }
    void display() const {
        if (size == 0) {
            cout << "List is empty." << endl;
            return;
        }
        for (int i = 0; i < size; i++) {
            cout << "Student " << i << ": Reg: " << students[i].get_regNo()
                 << ", Name: " << students[i].get_name()
                 << ", Age: " << students[i].get_age()
                 << ", Course: " << students[i].get_course().get_course_code()
                 << " (" << students[i].get_course().get_course_name() << ")"
                 << ", Mark: " << students[i].get_grade().get_mark()
                 << ", Grade: " << students[i].get_grade().get_grade() << endl;
        }
    }
};
