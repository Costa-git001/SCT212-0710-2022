#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_REG 20
#define MAX_NAME 50

struct Course {
    char course_code[20];
    char course_name[50];
};

struct Grade {
    int mark;
    char the_grade;
    int is_set;
};

struct Student {
    char reg_number[MAX_REG + 1];
    char name[MAX_NAME + 1];
    int age;
    struct Course course;
    struct Grade grade;
};

struct Node {
    struct Student student;
    struct Node* next;
};

struct List {
    struct Node* head;
    int size;
};

void course_init(struct Course* c, const char* code, const char* name) {
    strncpy(c->course_code, code, 19);
    c->course_code[19] = '\0';
    strncpy(c->course_name, name, 49);
    c->course_name[49] = '\0';
}

void course_set_code(struct Course* c, const char* code) {
    strncpy(c->course_code, code, 19);
    c->course_code[19] = '\0';
}

void course_set_name(struct Course* c, const char* name) {
    strncpy(c->course_name, name, 49);
    c->course_name[49] = '\0';
}

char* course_get_code(struct Course* c) {
    return c->course_code;
}

char* course_get_name(struct Course* c) {
    return c->course_name;
}

void grade_init(struct Grade* g, int m) {
    g->mark = m;
    g->is_set = 0;
    if (m >= 0 && m <= 100) {
        g->the_grade = m > 69 ? 'A' : m > 59 ? 'B' : m > 49 ? 'C' : m > 39 ? 'D' : 'E';
        g->is_set = 1;
    } else {
        g->the_grade = 'E';
    }
}

void grade_set_mark(struct Grade* g, int m) {
    if (!g->is_set && m >= 0 && m <= 100) {
        g->mark = m;
        g->the_grade = m > 69 ? 'A' : m > 59 ? 'B' : m > 49 ? 'C' : m > 39 ? 'D' : 'E';
        g->is_set = 1;
    }
}

int grade_get_mark(struct Grade* g) {
    return g->mark;
}

char grade_get_grade(struct Grade* g) {
    return g->the_grade;
}

int grade_is_set(struct Grade* g) {
    return g->is_set;
}

void student_init(struct Student* s, const char* reg, const char* name, int age, 
                 struct Course c, struct Grade g) {
    strncpy(s->reg_number, reg, MAX_REG);
    s->reg_number[MAX_REG] = '\0';
    strncpy(s->name, name, MAX_NAME);
    s->name[MAX_NAME] = '\0';
    s->age = age;
    s->course = c;
    s->grade = g;
}

void student_set_regNo(struct Student* s, const char* reg) {
    strncpy(s->reg_number, reg, MAX_REG);

}

void student_set_name(struct Student* s, const char* name) {

void student_set_age(struct Student* s, int age) {

void student_set_course(struct Student* s, struct Course c) {
    s->course = c;
}

void student_set_grade(struct Student* s, struct Grade g) {
    s->grade = g;
}

void student_set_marks(struct Student* s, int m) {
    grade_set_mark(&s->grade, m);
}

char* student_get_regNo(struct Student* s) {
    return s->reg_number;
}

char* student_get_name(struct Student* s) {
    return s->name;

int student_get_age(struct Student* s) {

    return s->grade;

int student_get_marks(struct Student* s) {
    return grade_get_mark(&s->grade);
}

void student_calculate_grade(struct Student* s) {
    if (!grade_is_set(&s->grade)) {
        grade_set_mark(&s->grade, grade_get_mark(&s->grade));
    } else {
        printf("Grade for %s is already set and cannot be modified.\n", s->reg_number);
    }
}

void list_init(struct List* list) {
    list->head = NULL;
    list->size = 0;
}

void list_init_single(struct List* list, struct Student s) {
    list->head = (struct Node*)malloc(sizeof(struct Node));
    list->head->student = s;
    list->head->next = NULL;
    list->size = 1;
}

void list_init_copy(struct List* list, struct List* other) {
    list->head = NULL;
    list->size = 0;
    struct Node* current = other->head;
    struct Node* last = NULL;
    while (current != NULL) {
        struct Node* new_node = (struct Node*)malloc(sizeof(struct Node));
        new_node->student = current->student;
        new_node->next = NULL;
        if (list->head == NULL) {
            list->head = new_node;
        } else {
            last->next = new_node;
        }
        last = new_node;
        list->size++;
        current = current->next;
    }
}

void list_add_student(struct List* list, struct Student s) {
    struct Node* new_node = (struct Node*)malloc(sizeof(struct Node));
    new_node->student = s;
    new_node->next = NULL;
    if (list->head == NULL) {
        list->head = new_node;
    } else {
        struct Node* current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
    list->size++;
    printf("Student %s added.\n", s.reg_number);
}

void list_add_student_at(struct List* list, struct Student s, int position) {
    if (position < 0 || position > list->size) {
        printf("Invalid position!\n");
        return;
    }
    struct Node* new_node = (struct Node*)malloc(sizeof(struct Node));
    new_node->student = s;
    if (position == 0) {
        new_node->next = list->head;
        list->head = new_node;
    } else {
        struct Node* current = list->head;
        for (int i = 0; i < position - 1; i++) {
            current = current->next;
        }
        new_node->next = current->next;
        current->next = new_node;
    }
    list->size++;
    printf("Student %s added at position %d.\n", s.reg_number, position);
}

    while (current != NULL) {
        if (strcmp(current->student.reg_number, s.reg_number) == 0) {
            if (prev == NULL) {

            free(current);
            list->size--;
            printf("Student %s removed.\n", s.reg_number);

void list_remove_student_at(struct List* list, int position) {
    if (position < 0 || position >= list->size || list->head == NULL) {
        printf("Invalid position!\n");
        return;
    }
    struct Node* current = list->head;
    struct Node* prev = NULL;
    for (int i = 0; i < position; i++) {
        prev = current;
        current = current->next;
    }
    if (prev == NULL) {
        list->head = current->next;
    } else {
        prev->next = current->next;
    }
    free(current);
    list->size--;
    printf("Student at position %d removed.\n", position);
}

int list_get_size(struct List* list) {
    return list->size;
}

void list_destroy(struct List* list) {
    struct Node* current = list->head;
    while (current != NULL) {
        struct Node* temp = current;
        current = current->next;
        free(temp);
    }
    list->head = NULL;
    list->size = 0;
    printf("List cleared.\n");
}

struct Student* list_get_student(struct List* list, int index) {
    if (index < 0 || index >= list->size) {
        printf("Invalid index!\n");
        return NULL;
    }
    struct Node* current = list->head;
    for (int i = 0; i < index; i++) {
        current = current->next;
    }
    return &current->student;
}

void list_display(struct List* list) {
    if (list->size == 0) {
        printf("List is empty.\n");
        return;
    }
    struct Node* current = list->head;
    int i = 0;
    while (current != NULL) {
        printf("Student %d: Reg: %s, Name: %s, Age: %d, Course: %s (%s), Mark: %d, Grade: %c\n",
               i, current->student.reg_number, current->student.name, current->student.age,
               current->student.course.course_code, current->student.course.course_name,
               current->student.grade.mark, current->student.grade.the_grade);
        current = current->next;
        i++;
    }
}
